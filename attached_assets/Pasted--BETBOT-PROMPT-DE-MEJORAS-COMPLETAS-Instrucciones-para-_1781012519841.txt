```python
BANKROLL_DEFAULT = 1000.0

def _load_live_bankroll() -> float:
    """
    Read the current bankroll from bankroll_log.csv (most recent entry).
    Falls back to BANKROLL_DEFAULT if no log exists yet.
    """
    try:
        if os.path.isfile(BANKROLL_LOG_FILE):
            import csv as _csv
            last_balance = BANKROLL_DEFAULT
            with open(BANKROLL_LOG_FILE, "r", newline="", encoding="utf-8") as _f:
                rows = list(_csv.DictReader(_f))
            if rows:
                # Find the most recent row with a 'balance' field
                for row in reversed(rows):
                    bal = float(row.get("balance") or row.get("bankroll") or 0)
                    if bal > 0:
                        last_balance = bal
                        break
            return last_balance
    except Exception as _e:
        print(f"  ⚠️  _load_live_bankroll error: {_e}")
    return BANKROLL_DEFAULT

BANKROLL = _load_live_bankroll()
```

```python
def _elo_to_goals(elo_home: float, elo_away: float) -> tuple:
    """
    Convert ELO ratings to expected goals per team using a log-ratio model.
    Based on WC historical scoring rates (avg ~1.3 goals/team/game).
    
    Returns (avg_goals_home, avg_goals_away).
    """
    BASE_GOALS = 1.30
    HOME_BOOST = 0.10          # slight venue advantage in WC neutral-site games
    ELO_SCALE  = 500.0         # controls sensitivity: ±500 ELO ≈ ±30% goals

    elo_ratio = (elo_home - elo_away) / ELO_SCALE

    avg_h = BASE_GOALS * (1.0 + elo_ratio) + HOME_BOOST
    avg_a = BASE_GOALS * (1.0 - elo_ratio)

    # Hard caps: realistic WC range
    avg_h = max(0.40, min(3.50, avg_h))
    avg_a = max(0.40, min(3.50, avg_a))

    return round(avg_h, 2), round(avg_a, 2)
```

## TASK 5 — NEW FEATURE: NRFI/YRFI MARKET (No/Yes Run First Inning)

```python
def analyze_nrfi(home_team: str, away_team: str,
                 hp_name: str, ap_name: str,
                 hp_era: float, ap_era: float,
                 home_ops: str, away_ops: str,
                 home_k9: float = 0.0, away_k9: float = 0.0) -> dict:
    """
    Analyze NRFI (No Run First Inning) / YRFI (Yes Run First Inning) market.
    
    Logic:
    - Both pitchers ERA < 3.00 → strong NRFI lean
    - Both pitchers ERA < 4.00 → NRFI lean
    - Either pitcher ERA > 5.00 → YRFI lean
    - K/9 > 9.5 for both → NRFI boost
    - OPS > .800 for either team → YRFI boost
    
    Returns:
      bet: "NRFI" | "YRFI" | "SKIP"
      confidence: "ALTA" | "MEDIA" | "BAJA"
      reason: explanation string in Spanish
      ev_est: estimated EV% at typical odds (-135 NRFI / +105 YRFI)
    """
    score = 0   # positive = NRFI, negative = YRFI

    reasons = []

    # Pitcher ERA scoring
    if hp_era < 3.00: score += 3; reasons.append(f"Lanzador local élite ({hp_era:.2f} prom. carreras)")
    elif hp_era < 4.00: score += 1; reasons.append(f"Lanzador local sólido ({hp_era:.2f})")
    elif hp_era > 5.00: score -= 2; reasons.append(f"⚠️ Lanzador local débil ({hp_era:.2f})")

    if ap_era < 3.00: score += 3; reasons.append(f"Lanzador visitante élite ({ap_era:.2f})")
    elif ap_era < 4.00: score += 1; reasons.append(f"Lanzador visitante sólido ({ap_era:.2f})")
    elif ap_era > 5.00: score -= 2; reasons.append(f"⚠️ Lanzador visitante débil ({ap_era:.2f})")

    # K/9 scoring (high K rate → batters struggle in 1st inning)
    if home_k9 > 9.5: score += 1; reasons.append(f"Lanzador local: muchos ponches/9 innings ({home_k9:.1f})")
    if away_k9 > 9.5: score += 1; reasons.append(f"Lanzador visitante: muchos ponches/9 innings ({away_k9:.1f})")

    # OPS scoring (high OPS → teams score in 1st inning more often)
    try:
        if float(home_ops or 0) > 0.800: score -= 1; reasons.append(f"Ofensiva local peligrosa ({home_ops})")
        if float(away_ops or 0) > 0.800: score -= 1; reasons.append(f"Ofensiva visitante peligrosa ({away_ops})")
    except Exception:
        pass

    # Decision
    if score >= 5:
        bet = "NRFI"; confidence = "ALTA"
        # Estimate EV at -135 NRFI odds (decimal 1.741) → true prob needed ~60%
        true_prob = 0.68
        ev_est = round((true_prob * 1.741 - 1) * 100, 1)
    elif score >= 2:
        bet = "NRFI"; confidence = "MEDIA"
        true_prob = 0.60
        ev_est = round((true_prob * 1.741 - 1) * 100, 1)
    elif score <= -3:
        bet = "YRFI"; confidence = "ALTA"
        # Typical YRFI odds +105 (decimal 2.05) → true prob ~50%
        true_prob = 0.52
        ev_est = round((true_prob * 2.05 - 1) * 100, 1)
    elif score <= -1:
        bet = "YRFI"; confidence = "MEDIA"
        true_prob = 0.48
        ev_est = round((true_prob * 2.05 - 1) * 100, 1)
    else:
        return {"bet": "SKIP", "confidence": "BAJA", "reason": "Sin ventaja clara en el primer inning.", "ev_est": 0.0}

    reason = " | ".join(reasons) if reasons else "Análisis de lanzadores"
    return {"bet": bet, "confidence": confidence, "reason": reason, "ev_est": ev_est}
```

## TASK 6 — IMPROVEMENT: CORRELATED PARLAY PROTECTION

When building auto-parlays, some market combinations are correlated and reduce the true value of the parlay.

**Action:** Add this validation function:

```python
_CORRELATED_PAIRS: list = [
    # (market_type_a, direction_a, market_type_b, direction_b, warning)
    ("strikeout_prop", "over",  "totals",    "under", "K alto + Under son correlacionados — menos carreras cuando el pitcher domina"),
    ("strikeout_prop", "under", "totals",    "over",  "K bajo + Over son correlacionados — más carreras cuando hay mal pitcheo"),
    ("h2h",           None,    "spreads",   None,    "Moneyline + línea de carreras del mismo equipo son redundantes"),
    ("h2h",           None,    "h2h_first5",None,    "Moneyline partido completo + primer tiempo comparten resultado"),
]

def _check_correlated_legs(legs: list) -> str:
    """
    Given a list of parlay leg dicts (each with 'market_type', 'team', 'direction'),
    return a warning string if any pair is correlated, or "" if all clear.
    """
    for i, leg_a in enumerate(legs):
        for leg_b in legs[i+1:]:
            for ma, da, mb, db, warn in _CORRELATED_PAIRS:
                a_match = (leg_a.get("market_type","") == ma and
                           (da is None or leg_a.get("direction","") == da))
                b_match = (leg_b.get("market_type","") == mb and
                           (db is None or leg_b.get("direction","") == db))
                if a_match and b_match:
                    return f"⚠️ PARLAY CORRELACIONADO: {warn}"
                # Check reverse order
                a_match2 = (leg_a.get("market_type","") == mb and
                            (db is None or leg_a.get("direction","") == db))
                b_match2 = (leg_b.get("market_type","") == ma and
                            (da is None or leg_b.get("direction","") == da))
                if a_match2 and b_match2:
                    return f"⚠️ PARLAY CORRELACIONADO: {warn}"
    return ""
```

In the parlay construction logic, call `_check_correlated_legs(legs)` before sending the parlay alert.

```python
def _check_reverse_line_movement(game_id: str, home_team: str, away_team: str,
                                  prev_ml_home: float, curr_ml_home: float,
                                  prev_total: float, curr_total: float) -> list:
    """
    Detect reverse line movement patterns.
    Returns list of alert strings (empty = no RLM detected).
    
    RLM logic (approximation without public % data):
    - If home team is a heavy favorite (ML < -200) and line SHORTENS further → public bet,
      but if line LENGTHENS (favorite pays more) → sharp fade signal.
    - If total moves DOWN despite home team being hitter-friendly park → sharp UNDER signal.
    """
    alerts = []

    # Moneyline RLM: heavy favorite getting worse odds (sharps fading)
    if prev_ml_home and curr_ml_home:
        if prev_ml_home < -180 and curr_ml_home > prev_ml_home + 0.05:
            # Favorite's odds lengthened — public on them but sharps fading
            alerts.append(
                f"🔄 MOVIMIENTO INVERSO (GANADOR): El público apuesta {_es(home_team)} "
                f"pero la línea se mueve en su contra. Posible dinero sharp en {_es(away_team)}."
            )

    # Total RLM: line moved DOWN in a hitter-friendly park
    if prev_total and curr_total:
        park_factor = MLB_PARK_FACTORS.get(home_team, 1.0)
        if park_factor > 1.05 and curr_total < prev_total - 0.3:
            alerts.append(
                f"🔄 MOVIMIENTO INVERSO (TOTAL): Estadio favorece bateadores "
                f"pero la línea bajó {prev_total} → {curr_total}. "
                f"Sharp money en UNDER."
            )

    return alerts
```
